cmake_minimum_required(VERSION 3.21)

if(CMAKE_SCRIPT_MODE_FILE)
    execute_process(COMMAND ${RC_MP_CHECK_NM} -g --defined-only -P ${RC_MP_CHECK_ARCHIVE}
                    RESULT_VARIABLE _rc_mp_rc OUTPUT_VARIABLE _rc_mp_out ERROR_VARIABLE _rc_mp_err)
    if(NOT _rc_mp_rc EQUAL 0 OR _rc_mp_out STREQUAL "")
        file(REMOVE ${RC_MP_CHECK_ARCHIVE})
        message(FATAL_ERROR "rayclay_mediaprofile: '${RC_MP_CHECK_NM}' failed (${_rc_mp_rc}) or listed nothing on "
                            "${RC_MP_CHECK_ARCHIVE}: ${_rc_mp_err}")
    endif()
    string(REPLACE "\n" ";" _rc_mp_lines "${_rc_mp_out}")
    set(_rc_mp_foreign "")
    set(_rc_mp_count 0)
    foreach(_rc_mp_line IN LISTS _rc_mp_lines)
        if(_rc_mp_line STREQUAL "" OR _rc_mp_line MATCHES ":$" OR _rc_mp_line MATCHES "^\\.refptr\\.")
            continue()
        endif()
        string(REGEX MATCH "^[^ ]+" _rc_mp_name "${_rc_mp_line}")
        if(_rc_mp_name MATCHES "^_?___asan_globals_registered$")
            continue()
        endif()
        if(_rc_mp_name MATCHES "^(_?)__odr_asan(\\.|_gen_)(.+)$")
            set(_rc_mp_name "${CMAKE_MATCH_1}${CMAKE_MATCH_3}")
        endif()
        math(EXPR _rc_mp_count "${_rc_mp_count} + 1")
        if(NOT _rc_mp_name MATCHES "^_?rci_")
            list(APPEND _rc_mp_foreign "${_rc_mp_name}")
        endif()
    endforeach()
    if(_rc_mp_foreign)
        list(LENGTH _rc_mp_foreign _rc_mp_n)
        if(_rc_mp_n GREATER 8)
            list(SUBLIST _rc_mp_foreign 0 8 _rc_mp_foreign)
            list(APPEND _rc_mp_foreign "...")
        endif()
        file(REMOVE ${RC_MP_CHECK_ARCHIVE})
        message(FATAL_ERROR "rayclay_mediaprofile: the archive defines ${_rc_mp_n} global(s) outside rci_, which an "
                            "application's own copy of a decoder could meet: ${_rc_mp_foreign} (add them with "
                            "tools/media-profile-webp-names.py [--lib nestegg|opus] or media-profile-vpx-names.py "
                            "--update <the objects>)")
    endif()
    message(STATUS "rayclay_mediaprofile: the archive's ${_rc_mp_count} globals are all RayClay's (rci_)")
    return()
endif()

if(NOT DEFINED RC_MEDIAPROFILE_DIR)
    set(RC_MEDIAPROFILE_DIR "${CMAKE_CURRENT_SOURCE_DIR}")
endif()
if(NOT DEFINED RC_MEDIAPROFILE_AUDIO)
    if(CMAKE_SYSTEM_NAME STREQUAL "Linux")
        set(RC_MEDIAPROFILE_AUDIO ON)
    else()
        set(RC_MEDIAPROFILE_AUDIO OFF)
    endif()
endif()
if(NOT DEFINED RC_MEDIAPROFILE_WEBP)
    if(APPLE)
        set(RC_MEDIAPROFILE_WEBP OFF)
    else()
        set(RC_MEDIAPROFILE_WEBP ON)
    endif()
endif()
if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/rc_image_webp_profile.c")
    set(_rc_mp_webp_src "${CMAKE_CURRENT_SOURCE_DIR}")
else()
    get_filename_component(_rc_mp_webp_src "${CMAKE_CURRENT_SOURCE_DIR}/../../gui/webp" ABSOLUTE)
endif()
set(_rc_mp_need PINS)
if(RC_MEDIAPROFILE_AUDIO)
    list(APPEND _rc_mp_need dr_mp3.h dr_flac.h stb_vorbis.c libvpx/LICENSE libvpx/vpx/vpx_decoder.h
         nestegg/src/nestegg.c opus/COPYING)
endif()
if(RC_MEDIAPROFILE_WEBP)
    list(APPEND _rc_mp_need libwebp/COPYING libwebp/src/webp/decode.h libwebp/src/webp/demux.h)
endif()
foreach(_rc_mp_file IN LISTS _rc_mp_need)
    if(NOT EXISTS "${RC_MEDIAPROFILE_DIR}/${_rc_mp_file}")
        message(FATAL_ERROR "rayclay: the media profile lacks ${_rc_mp_file} at RC_MEDIAPROFILE_DIR="
                            "${RC_MEDIAPROFILE_DIR}")
    endif()
endforeach()
if(NOT RC_MEDIAPROFILE_AUDIO AND NOT RC_MEDIAPROFILE_WEBP)
    message(FATAL_ERROR "rayclay: the media profile is added with neither RC_MEDIAPROFILE_AUDIO nor "
                        "RC_MEDIAPROFILE_WEBP")
endif()

add_library(rayclay_mediaprofile STATIC)
set_target_properties(rayclay_mediaprofile PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS ON
                      POSITION_INDEPENDENT_CODE ON)
set(_rc_mp_sections "")
if(MINGW)
    set(_rc_mp_sections -ffunction-sections)
elseif(CMAKE_C_COMPILER_ID MATCHES "GNU|Clang")
    set(_rc_mp_sections -ffunction-sections -fdata-sections)
endif()
target_compile_options(rayclay_mediaprofile PRIVATE ${_rc_mp_sections})
if(UNIX AND NOT APPLE)
    target_link_libraries(rayclay_mediaprofile PUBLIC m)
endif()

if(RC_MEDIAPROFILE_AUDIO)
    target_sources(rayclay_mediaprofile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/rc_media_profile.c")
    target_include_directories(rayclay_mediaprofile SYSTEM PRIVATE "${RC_MEDIAPROFILE_DIR}")
    target_include_directories(rayclay_mediaprofile PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")

    set(_rc_mp_vpx_root "${RC_MEDIAPROFILE_DIR}/libvpx")
    file(GLOB_RECURSE _rc_mp_vpx_c LIST_DIRECTORIES false "${_rc_mp_vpx_root}/*.c")
    list(SORT _rc_mp_vpx_c)
    add_library(rayclay_mediaprofile_libvpx OBJECT ${_rc_mp_vpx_c})
    set_target_properties(rayclay_mediaprofile_libvpx PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON
                          C_EXTENSIONS ON POSITION_INDEPENDENT_CODE ON)
    target_include_directories(rayclay_mediaprofile_libvpx PRIVATE "${_rc_mp_vpx_root}")
    target_compile_options(rayclay_mediaprofile_libvpx PRIVATE ${_rc_mp_sections} -w
                           -include "${CMAKE_CURRENT_SOURCE_DIR}/rc_mp_vpx_names.h")
    target_sources(rayclay_mediaprofile PRIVATE $<TARGET_OBJECTS:rayclay_mediaprofile_libvpx>)
    target_include_directories(rayclay_mediaprofile SYSTEM PRIVATE "${_rc_mp_vpx_root}")
    find_package(Threads REQUIRED)
    target_link_libraries(rayclay_mediaprofile PUBLIC Threads::Threads)
    set(RC_MEDIAPROFILE_VPX_SOURCES ${_rc_mp_vpx_c} PARENT_SCOPE)
    set(RC_MEDIAPROFILE_VPX_INCLUDE "${_rc_mp_vpx_root}" PARENT_SCOPE)
    set(RC_MEDIAPROFILE_VPX_NAMES "${CMAKE_CURRENT_SOURCE_DIR}/rc_mp_vpx_names.h" PARENT_SCOPE)
endif()

if(RC_MEDIAPROFILE_AUDIO)
    set(_rc_mp_nestegg "${RC_MEDIAPROFILE_DIR}/nestegg")
    set(_rc_mp_opus "${RC_MEDIAPROFILE_DIR}/opus")
    file(GLOB _rc_mp_opus_c LIST_DIRECTORIES false
         "${_rc_mp_opus}/src/*.c" "${_rc_mp_opus}/celt/*.c" "${_rc_mp_opus}/silk/*.c")
    list(SORT _rc_mp_opus_c)
    add_library(rayclay_mediaprofile_nestegg OBJECT "${_rc_mp_nestegg}/src/nestegg.c")
    add_library(rayclay_mediaprofile_opus OBJECT ${_rc_mp_opus_c})
    target_include_directories(rayclay_mediaprofile_nestegg PRIVATE "${_rc_mp_nestegg}/include")
    target_include_directories(rayclay_mediaprofile_opus PRIVATE "${_rc_mp_opus}/include" "${_rc_mp_opus}/celt"
                               "${_rc_mp_opus}/silk")
    if(MSVC)
        set(_rc_mp_opus_stack USE_ALLOCA)
    else()
        set(_rc_mp_opus_stack VAR_ARRAYS)
    endif()
    target_compile_definitions(rayclay_mediaprofile_opus PRIVATE OPUS_BUILD ${_rc_mp_opus_stack} ENABLE_HARDENING
                               HAVE_LRINTF PACKAGE_VERSION="1.6.1")
    foreach(_rc_mp_lib nestegg opus)
        set(_rc_mp_obj rayclay_mediaprofile_${_rc_mp_lib})
        set(_rc_mp_lib_names "${CMAKE_CURRENT_SOURCE_DIR}/rc_mp_${_rc_mp_lib}_names.h")
        if(MSVC)
            target_compile_options(${_rc_mp_obj} PRIVATE ${_rc_mp_sections} /W0 "/FI${_rc_mp_lib_names}")
        else()
            target_compile_options(${_rc_mp_obj} PRIVATE ${_rc_mp_sections} -w -include "${_rc_mp_lib_names}")
        endif()
        set_target_properties(${_rc_mp_obj} PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON C_EXTENSIONS ON
                              POSITION_INDEPENDENT_CODE ON)
        target_sources(rayclay_mediaprofile PRIVATE $<TARGET_OBJECTS:${_rc_mp_obj}>)
    endforeach()
    target_include_directories(rayclay_mediaprofile SYSTEM PRIVATE "${_rc_mp_nestegg}/include" "${_rc_mp_opus}/include")
endif()

if(RC_MEDIAPROFILE_WEBP)
    set(_rc_mp_webp_root "${RC_MEDIAPROFILE_DIR}/libwebp")
    file(GLOB _rc_mp_webp_c LIST_DIRECTORIES false
         "${_rc_mp_webp_root}/src/dec/*.c" "${_rc_mp_webp_root}/src/demux/*.c"
         "${_rc_mp_webp_root}/src/dsp/*.c" "${_rc_mp_webp_root}/src/utils/*.c")
    list(SORT _rc_mp_webp_c)
    set(_rc_mp_names "${_rc_mp_webp_src}/rc_image_webp_names.h")
    if(MSVC)
        set(_rc_mp_webp_flags /W0 "/FI${_rc_mp_names}")
    else()
        set(_rc_mp_webp_flags -w -include "${_rc_mp_names}")
    endif()
    set(_rc_mp_webp_defs WEBP_MAX_ALLOCABLE_MEMORY=201326592ULL WEBP_REDUCE_SIZE WEBP_REDUCE_CSP)
    add_library(rayclay_mediaprofile_libwebp OBJECT ${_rc_mp_webp_c})
    set_target_properties(rayclay_mediaprofile_libwebp PROPERTIES C_STANDARD 99 C_STANDARD_REQUIRED ON
                          C_EXTENSIONS ON POSITION_INDEPENDENT_CODE ON)
    target_include_directories(rayclay_mediaprofile_libwebp PRIVATE "${_rc_mp_webp_root}")
    target_compile_definitions(rayclay_mediaprofile_libwebp PRIVATE "${_rc_mp_webp_defs}")
    target_compile_options(rayclay_mediaprofile_libwebp PRIVATE ${_rc_mp_sections} ${_rc_mp_webp_flags})
    target_sources(rayclay_mediaprofile PRIVATE $<TARGET_OBJECTS:rayclay_mediaprofile_libwebp>)
    set(_rc_mp_bridge "${_rc_mp_webp_src}/rc_image_webp_profile.c")
    target_sources(rayclay_mediaprofile PRIVATE "${_rc_mp_bridge}")
    target_include_directories(rayclay_mediaprofile SYSTEM PRIVATE "${_rc_mp_webp_root}")
    target_include_directories(rayclay_mediaprofile PRIVATE "${_rc_mp_webp_src}")
    if(MSVC)
        set(_rc_mp_bridge_fi "/FI${_rc_mp_names}")
    else()
        set(_rc_mp_bridge_fi -include "${_rc_mp_names}")
    endif()
    set_source_files_properties("${_rc_mp_bridge}" PROPERTIES COMPILE_OPTIONS "${_rc_mp_bridge_fi}"
                                COMPILE_DEFINITIONS "${_rc_mp_webp_defs}")
    set(RC_MEDIAPROFILE_WEBP_SOURCES ${_rc_mp_webp_c} "${_rc_mp_bridge}" PARENT_SCOPE)
    set(RC_MEDIAPROFILE_WEBP_INCLUDE "${_rc_mp_webp_root}" "${_rc_mp_webp_src}" PARENT_SCOPE)
endif()

                -DRC_MP_CHECK_ARCHIVE=$<TARGET_FILE:rayclay_mediaprofile> -P ${CMAKE_CURRENT_LIST_FILE}
        VERBATIM)
endif()
